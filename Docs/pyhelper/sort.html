26    """
27    Sort *seq* in place using a straight insertion sort.
28
29    Insertion sort builds the final sorted array one element at a time.
30    It is efficient for small sequences or nearly sorted data.
31
32    ==============================================
33    Insertion Sort
34    Best Time complexity: O(n)
35    Avg Time complexity: O(n²)
36    Worst Time complexity: O(n²)
37    Space complexity: O(1)   (in‑place)
38    Stable: Yes
39    ==============================================
40
41    Args:
42        seq: The sequence to be sorted in place.
43        key: A function that extracts a comparison key from each element.
44             Default is the identity function.
45        reverse: Whether to sort in descending order (default ``False``).
46
47    Raises:
48        TypeError: If elements (after applying *key*) are not comparable.
49
50    Examples:
51        >>> data = [5, 2, 8, 1, 9, 3]
52        >>> insertion_sort(data)
53        >>> data
54        [1, 2, 3, 5, 8, 9]
55
56        >>> words = ['banana', 'apple', 'cherry', 'date']
57        >>> insertion_sort(words, key=len)
58        >>> words
59        ['date', 'apple', 'banana', 'cherry']
60
61        >>> nums = [3, 1, 4, 1, 5, 9]
62        >>> insertion_sort(nums, reverse=True)
63        >>> nums
64        [9, 5, 4, 3, 1, 1]
65    """
66    # Define a local key function (identity if none provided)
67    kf = key if key is not None else (lambda x: x)
68
69    n = len(seq)
70    # Outer loop: take each element as the "current" one to insert
71    for i in range(1, n):
72        current = seq[i]          # element to be placed correctly
73        cur_key = kf(current)
74        j = i - 1
75
76        # Shift elements to the right while they are greater than current
77        # (or smaller if reverse is True)
78        while j >= 0:
79            cmp = kf(seq[j]) > cur_key if not reverse else kf(seq[j]) < cur_key
80            if not cmp:
81                break
82            seq[j + 1] = seq[j]
83            j -= 1
84
85        seq[j + 1] = current      # insert current at its correct position

160        seq: MutableSequence,
161        key: Optional[Callable[[Any], Any]] = None,
162        reverse: bool = False,
163) -> None:
164    """
165    Sort *seq* in place using a bottom‑up iterative merge sort.
166
167    This implementation uses a temporary buffer of the same length as
168    *seq* to avoid recursive overhead and to guarantee stability.
169    The merge step compares elements using the supplied *key*.
170
171    ==============================================
172    Merge Sort (iterative, bottom‑up)
173    Best Time complexity: O(n log n)
174    Avg Time complexity: O(n log n)
175    Worst Time complexity: O(n log n)
176    Space complexity: O(n)   (auxiliary buffer)
177    Stable: Yes
178    ==============================================
179
180    Args:
181        seq: The sequence to be sorted in place.
182        key: A function that extracts a comparison key from each element.
183        reverse: Whether to sort in descending order.
184
185    Raises:
186        TypeError: If elements (after *key*) are not comparable.
187
188    Examples:
189        >>> data = [5, 2, 8, 1, 9, 3]
190        >>> merge_sort(data)
191        >>> data
192        [1, 2, 3, 5, 8, 9]
193
194        >>> words = ['banana', 'apple', 'cherry', 'date']
195        >>> merge_sort(words, key=len)
196        >>> words
197        ['date', 'apple', 'banana', 'cherry']
198    """
199    kf = key if key is not None else (lambda x: x)
200    n = len(seq)
201    # Temporary buffer for merging – reuse it across all merges
202    aux = [None] * n
203
204    width = 1
205    while width < n:
206        left = 0
207        while left < n:
208            mid = left + width
209            if mid >= n:
210                break          # no right half to merge
211            right = min(left + 2 * width, n)
212            # Merge seq[left:mid] and seq[mid:right] into aux[left:right]
213            i, j, k = left, mid, left
214            while i < mid and j < right:
215                # Determine which element should come first
216                if not reverse:
217                    take_left = kf(seq[i]) <= kf(seq[j])
218                else:
219                    take_left = kf(seq[i]) >= kf(seq[j])
220                if take_left:
221                    aux[k] = seq[i]
222                    i += 1
223                else:
224                    aux[k] = seq[j]
225                    j += 1
226                k += 1
227            # Copy remaining elements from left half (if any)
228            while i < mid:
229                aux[k] = seq[i]
230                i += 1
231                k += 1
232            # Copy remaining elements from right half (if any)
233            while j < right:
234                aux[k] = seq[j]
235                j += 1
236                k += 1
237            # Copy merged segment back to original sequence
238            for p in range(left, right):
239                seq[p] = aux[p]
240            left = right
241        width *= 2

def quick_sort( seq: MutableSequence, key: Callable[[Any], Any] | None = None, reverse: bool = False) -> None: 
247def quick_sort(
248        seq: MutableSequence,
249        key: Optional[Callable[[Any], Any]] = None,
250        reverse: bool = False,
251) -> None:
252    """
253    Sort *seq* in place using the Quicksort algorithm (Lomuto partition).
254
255    This implementation uses an explicit stack to avoid recursion depth
256    issues. The pivot is chosen as the median of three (first, middle,
257    last). The algorithm is NOT stable.
258
259    ==============================================
260    Quick Sort (Lomuto partition, iterative)
261    Best Time complexity: O(n log n)
262    Avg Time complexity: O(n log n)
263    Worst Time complexity: O(n²)   (rare with median‑of‑three)
264    Space complexity: O(log n)     (stack for sub‑arrays)
265    Stable: No
266    ==============================================
267
268    Args:
269        seq: The sequence to be sorted in place.
270        key: A function that extracts a comparison key from each element.
271        reverse: Whether to sort in descending order.
272
273    Raises:
274        TypeError: If elements (after *key*) are not comparable.
275
276    Examples:
277        >>> data = [5, 2, 8, 1, 9, 3]
278        >>> quick_sort(data)
279        >>> data
280        [1, 2, 3, 5, 8, 9]
281
282        >>> nums = [3, 1, 4, 1, 5, 9]
283        >>> quick_sort(nums, reverse=True)
284        >>> nums
285        [9, 5, 4, 3, 1, 1]
286    """
287    kf = key if key is not None else (lambda x: x)
288    n = len(seq)
289    if n <= 1:
290        return
291
292    # Use a stack of (low, high) ranges to process
293    stack = [(0, n - 1)]
294
295    while stack:
296        low, high = stack.pop()
297        if low >= high:
298            continue
299
300        # Median-of-three pivot selection
301        mid = (low + high) // 2
302        # Sort the three candidates: seq[low], seq[mid], seq[high]
303        # Ascending:  seq[low] <= seq[mid] <= seq[high]
304        # Descending: seq[low] >= seq[mid] >= seq[high]
305        if not reverse:
306            if kf(seq[low]) > kf(seq[mid]):
307                seq[low], seq[mid] = seq[mid], seq[low]
308            if kf(seq[low]) > kf(seq[high]):
309                seq[low], seq[high] = seq[high], seq[low]
310            if kf(seq[mid]) > kf(seq[high]):
311                seq[mid], seq[high] = seq[high], seq[mid]
312        else:
313            if kf(seq[low]) < kf(seq[mid]):
314                seq[low], seq[mid] = seq[mid], seq[low]
315            if kf(seq[low]) < kf(seq[high]):
316                seq[low], seq[high] = seq[high], seq[low]
317            if kf(seq[mid]) < kf(seq[high]):
318                seq[mid], seq[high] = seq[high], seq[mid]
319        # Place pivot at end (high) after median-of-three
320        pivot_val = seq[mid]
321        pivot_key = kf(pivot_val)
322        # Swap pivot with the last element (we already know high is >= pivot)
323        seq[mid], seq[high] = seq[high], seq[mid]
324
325        # Lomuto partition: rearrange around pivot
326        i = low - 1
327        for j in range(low, high):
328            # Decide if seq[j] belongs to the left side
329            if not reverse:
330                cond = kf(seq[j]) <= pivot_key
331            else:
332                cond = kf(seq[j]) >= pivot_key
333            if cond:
334                i += 1
335                seq[i], seq[j] = seq[j], seq[i]
336        # Place pivot in its final position
337        seq[i + 1], seq[high] = seq[high], seq[i + 1]
338        pivot_pos = i + 1
339
340        # Push sub‑ranges onto stack (larger range first to limit stack depth)
341        if pivot_pos - 1 - low > high - pivot_pos - 1:
342            stack.append((low, pivot_pos - 1))
343            stack.append((pivot_pos + 1, high))
344        else:
345            stack.append((pivot_pos + 1, high))
346            stack.append((low, pivot_pos - 1))

def heap_sort( seq: MutableSequence, key: Callable[[Any], Any] | None = None, reverse: bool = False) -> None: 
352def heap_sort(
353        seq: MutableSequence,
354        key: Optional[Callable[[Any], Any]] = None,
355        reverse: bool = False,
356) -> None:
357    """
358    Sort *seq* in place using the Heapsort algorithm.
359
360    Heapsort first builds a max‑heap (or min‑heap if *reverse* is True)
361    from the sequence, then repeatedly extracts the root to obtain a
362    sorted order. The algorithm is NOT stable.
363
364    ==============================================
365    Heap Sort
366    Best Time complexity: O(n log n)
367    Avg Time complexity: O(n log n)
368    Worst Time complexity: O(n log n)
369    Space complexity: O(1)   (in‑place)
370    Stable: No
371    ==============================================
372
373    Args:
374        seq: The sequence to be sorted in place.
375        key: A function that extracts a comparison key from each element.
376        reverse: Whether to sort in descending order.
377
378    Raises:
379        TypeError: If elements (after *key*) are not comparable.
380
381    Examples:
382        >>> data = [5, 2, 8, 1, 9, 3]
383        >>> heap_sort(data)
384        >>> data
385        [1, 2, 3, 5, 8, 9]
386
387        >>> nums = [3, 1, 4, 1, 5, 9]
388        >>> heap_sort(nums, reverse=True)
389        >>> nums
390        [9, 5, 4, 3, 1, 1]
391    """
392    kf = key if key is not None else (lambda x: x)
393    n = len(seq)
394
395    # Build heap (max‑heap for ascending, min‑heap for descending)
396    # Sift down operation for node i
397    def sift_down(start: int, end: int) -> None:
398        """Restore heap property for subtree rooted at *start*, up to *end*."""
399        root = start
400        while True:
401            child = 2 * root + 1
402            if child > end:
403                break
404            # Choose the child that should be swapped with parent
405            if not reverse:
406                # For max‑heap: pick larger child
407                if child + 1 <= end and kf(seq[child]) < kf(seq[child + 1]):
408                    child += 1
409                swap_cond = kf(seq[root]) < kf(seq[child])
410            else:
411                # For min‑heap: pick smaller child
412                if child + 1 <= end and kf(seq[child]) > kf(seq[child + 1]):
413                    child += 1
414                swap_cond = kf(seq[root]) > kf(seq[child])
415
416            if not swap_cond:
417                break
418            seq[root], seq[child] = seq[child], seq[root]
419            root = child
420
421    # Phase 1: build heap (start from last internal node)
422    for i in range(n // 2 - 1, -1, -1):
423        sift_down(i, n - 1)
424
425    # Phase 2: extract elements one by one
426    for i in range(n - 1, 0, -1):
427        seq[0], seq[i] = seq[i], seq[0]   # move current root to end
428        sift_down(0, i - 1)               # restore heap on reduced range

637def bucket_sort(
638        seq: MutableSequence,
639        key: Optional[Callable[[Any], float]] = None,
640        reverse: bool = False,
641) -> None:
642    """
643    Sort *seq* in place using the Bucket Sort algorithm.
644
645    **Important:** This algorithm distributes elements into buckets based on
646    their keys. By default it expects floating‑point numbers in **[0, 1)**.
647    If the keys fall outside this range, they are automatically normalized to
648    [0, 1) using min-max scaling — this means you can also use arbitrary
649    numeric keys (e.g. ``key=len`` for string length).
650
651    The algorithm distributes elements into buckets, sorts each bucket
652    individually (using insertion sort here), and then concatenates them.
653    It is stable only if the bucket sort used inside is stable (insertion
654    sort is stable).
655
656    ==============================================
657    Bucket Sort
658    Best Time complexity: O(n + k)   (uniform distribution)
659    Avg Time complexity: O(n + n²/k + k)   ≈ O(n) when k ~ n
660    Worst Time complexity: O(n²)   (all elements in one bucket)
661    Space complexity: O(n + k)
662    Stable: Yes (when inner sort is stable)
663    ==============================================
664
665    Args:
666        seq: The sequence to be sorted in place.
667        key: A function returning a comparable numeric value. Values are
668            automatically normalized to [0, 1) if necessary. Default is
669            identity.
670        reverse: Whether to sort in descending order.
671
672    Examples:
673        >>> data = [0.78, 0.17, 0.39, 0.26, 0.72, 0.94, 0.21, 0.12, 0.23, 0.68]
674        >>> bucket_sort(data)
675        >>> data
676        [0.12, 0.17, 0.21, 0.23, 0.26, 0.39, 0.68, 0.72, 0.78, 0.94]
677
678        >>> nums = [0.42, 0.32, 0.33, 0.52, 0.37, 0.47, 0.51]
679        >>> bucket_sort(nums, reverse=True)
680        >>> nums
681        [0.52, 0.51, 0.47, 0.42, 0.37, 0.33, 0.32]
682
683        >>> words = ['apple', 'bananas', 'cherry', 'date']
684        >>> bucket_sort(words, reverse=True, key=len)
685        >>> words
686        ['bananas', 'cherry', 'apple', 'date']
687    """
688    kf = key if key is not None else (lambda x: x)
689    n = len(seq)
690    if n <= 1:
691        return
692
693    # Collect raw keys
694    raw_keys = [kf(e) for e in seq]
695
