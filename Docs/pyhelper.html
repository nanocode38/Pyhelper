149def file_reopen(file_obj, stream=sys.stdout) -> Generator[None, Any, None]:
150    """
151    Context Manager: Temporarily change the standard output stream to the specified file.
152
153    Args:
154        file_obj: The Object of the file to redirect the standard output stream to.
155        stream: The stream to redirect.
156
157    Returns:
158        The original standard output stream.
159
160    Examples:
161        >>> original_stdin = sys.stdin
162        >>> original_stdout = sys.stdout
163        >>> if not os.path.isfile("test.in"):
164        ...     os.chdir("../tests")
165        >>> with open("test.in", "r", encoding="utf-8") as fb:
166        ...     with file_reopen(fb, "stdin"):
167        ...         print(sys.stdin == fb)
168        ...         file_input = input()
169        True
170        >>> sys.stdin == original_stdin
171        True
172        >>> file_input == "Hello, World!"
173        True
174        >>> with open("test.out", "w", encoding="utf-8") as fb:
175        ...     with file_reopen(fb, "stdout"):
176        ...         print("Hello, World!")
177        ...         spam = (sys.stdout == fb)
178        >>> sys.stdout == original_stdout
179        True
180        >>> spam
181        True
182        >>> with open("test.out", "r", encoding="utf-8") as fb:
183        ...     fb.read() == "Hello, World!\\n"
184        True
185        >>> with open("test.out", "w", encoding="utf-8"):
186        ...     pass
187    """
188    original_stream = sys.stdin
189    if isinstance(stream, str):
190        stream = stream.lower()
191    if stream in (sys.stdin, "stdin"):
192        sys.stdin = file_obj
193        yield
194        sys.stdin = original_stream
195    elif stream in (sys.stdout, "stdout"):
196        original_stream = sys.stdout
197        sys.stdout = file_obj
198        yield
199        sys.stdout = original_stream
200    elif stream in (sys.stderr, "stderr"):
201        original_stream = sys.stderr
202        sys.stderr = file_obj
203        yield
204        sys.stderr = original_stream
205    else:
206        raise ValueError("Invalid stream specified")

209def join_startup(target: Path, *args, **kwargs) -> bool:
210    """
211    Add a file to startup on Windows, macOS, or Linux.
212
213    Args:
214        target: Absolute path to the file/script to run at startup.
215        args and kwargs: Used to be compatible with old versions of name parameters
216
217    Returns:
218        True if successful, False otherwise.
219
220    Raises:
221        OSError: If the platform is not supported.
222
223    Notes:
224        - Windows: Uses registry (HKCU) for user-level startup
225        - macOS: Creates Launch Agent plist in ~/Library/LaunchAgents
226        - Linux: Creates systemd user service or .desktop file
227    """
228    # Prompts the user that the args and kwargs parameters have been abandoned.
229    # It is recommended to use the target parameter directly.
230    if args or kwargs:
231        import warnings
232
233        warnings.warn(
234            "args and kwargs parameters are deprecated and will be removed in a future version.",
235            DeprecationWarning,
236            stacklevel=2,
237        )
238
239    # Convert to absolute path and verify existence
240    target = os.path.abspath(target)
241    if not os.path.exists(target):
242        print(f"Error: File not found at {target}", file=sys.stderr)
243        return False
244
245    os_type: str = platform.system()
246    try:
247        if os_type == "Windows":
248            return _windows_startup(target)
249        elif os_type == "Darwin":  # macOS
250            return _macos_startup(target)
251        elif os_type == "Linux":
252            return _linux_startup(target)
253        else:
254            raise OSError("Unsupported platform, join_startup() is only available for Windows, MacOS and Linux systems")
255    except Exception as e:
256        print(f"Setup failed: {str(e)}")
257        return False

260def _windows_startup(file_path: str) -> bool:
261    """Windows implementation using registry"""
262    import winreg  # Standard library for registry access
263
264    # Determine execution command
265    cmd = f'"{file_path}"'  # Default for executables/batch files
266    if file_path.endswith(".py"):
267        python_exe = f'"{sys.executable}"'  # Use current Python interpreter
268        cmd = f'{python_exe} "{file_path}"'
269
270    # Create registry entry
271    key_path = r"Software\Microsoft\Windows\CurrentVersion\Run"
272    entry_name = "Startup_" + os.path.basename(file_path).replace(" ", "_")[:30]
273
274    try:
275        with winreg.OpenKey(winreg.HKEY_CURRENT_USER, key_path, 0, winreg.KEY_WRITE) as key:
276            winreg.SetValueEx(key, entry_name, 0, winreg.REG_SZ, cmd)
277        print(f"Added to startup: HKCU\\{key_path}\\{entry_name}")
278        return True
279    except WindowsError as e:
280        print(f"Registry error: {str(e)}")
281        return False

284def _macos_startup(file_path: str) -> bool:
285    """macOS implementation using LaunchAgent"""
286    plist_content = f"""<?xml version="1.0" encoding="UTF-8"?>
287<!DOCTYPE plist PUBLIC "-//Apple//DTD PLIST 1.0//EN" "https://www.apple.com/DTDs/PropertyList-1.0.dtd">
288<plist version="1.0">
289<dict>
290    <key>Label</key>
291    <string>com.startup.{os.path.basename(file_path)}</string>
292    <key>ProgramArguments</key>
293    <array>
294        <string>{sys.executable if file_path.endswith('.py') else '/bin/sh'}</string>
295        <string>{file_path}</string>
296    </array>
297    <key>RunAtLoad</key>
298    <true/>
299    <key>StandardOutPath</key>
300    <string>/tmp/{os.path.basename(file_path)}.log</string>
301    <key>StandardErrorPath</key>
302    <string>/tmp/{os.path.basename(file_path)}_error.log</string>
303</dict>
304</plist>"""
305
306    # Create LaunchAgents directory if missing
307    launch_agents_dir = os.path.expanduser("~/Library/LaunchAgents")
308    os.makedirs(launch_agents_dir, exist_ok=True)
309
310    # Write plist file
311    plist_name = f"com.startup.{Path(file_path).stem}.plist"
312    plist_path = os.path.join(launch_agents_dir, plist_name)
313
314    with open(plist_path, "w") as f:
315        f.write(plist_content)
316
317    # Load the agent
318    subprocess.run(["launchctl", "load", plist_path], check=True)
319    print(f"LaunchAgent created at {plist_path}")
320    return True

323def _linux_startup(file_path: str) -> bool:
324    """Linux implementation using systemd user service"""
325    service_content = f"""[Unit]
326Description=Startup Service: {os.path.basename(file_path)}
327After=network.target
328
329[Service]
330ExecStart={'/usr/bin/python3 ' if file_path.endswith('.py') else ''}{file_path}
331Restart=on-failure
332Environment="DISPLAY=:0"  # Required for GUI apps
333
334[Install]
335WantedBy=default.target
336"""
337
338    # Create systemd user directory
339    user_service_dir = os.path.expanduser("~/.config/systemd/user")
340    os.makedirs(user_service_dir, exist_ok=True)
341
342    # Write service file
343    service_name = f"startup_{Path(file_path).stem}.service"
344    service_path = os.path.join(user_service_dir, service_name)
345
346    with open(service_path, "w") as f:
347        f.write(service_content)
348
349    # Enable and start service
350    subprocess.run(["systemctl", "--user", "daemon-reload"], check=True)
351    subprocess.run(["systemctl", "--user", "enable", service_name], check=True)
352    subprocess.run(["systemctl", "--user", "start", service_name], check=True)
353
354    print(f"Systemd service created at {service_path}")
355    return True

392class Singleton(ABC):
393    """
394    An abstract base class to allow its subclass to be instantiated only once.
395
396    Warning:
397        If the subclass overloads the __new__() method, the parent class's __new__()
398        method must be called in the __new__() method of the subclass, otherwise this abstract base class is invalid
399
400    Examples:
401        >>> class FooSingleton(Singleton):
402        ...     def __init__(self):
403        ...         self.foo = 1
404        ...
405        >>> spam = FooSingleton()
406        >>> egg = FooSingleton()
407        Traceback (most recent call last):
408        ...
409        RuntimeError: The Singleton Class can only be instantiated once
410    """
411
412    _has_instantiation = False
413
414    def __new__(cls, *args, **kwargs):
415        if cls._has_instantiation:
416            raise RuntimeError("The Singleton Class can only be instantiated once")
417        cls._has_instantiation = True
418        return super().__new__(cls)

518    @functools.wraps(origin)
519    def __getattr__(self, name):

521            return kwargs[name]
